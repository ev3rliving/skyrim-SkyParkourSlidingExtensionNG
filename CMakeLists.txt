cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0141)
    cmake_policy(SET CMP0141 NEW)
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
endif()
project(MomentumSlideSE VERSION 1.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use the x64 Microsoft C++ toolchain.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
include(cmake/ModernDependencies.cmake)
add_library(MomentumSlideSE SHARED src/main.cpp)
target_link_libraries(MomentumSlideSE PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(MomentumSlideSE PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MomentumSlideSE PRIVATE /W4 /permissive- /utf-8)
enable_testing()
add_executable(SlidePhysicsTests tests/SlidePhysicsTests.cpp)
target_include_directories(SlidePhysicsTests PRIVATE src)
add_test(NAME SlidePhysics COMMAND SlidePhysicsTests)
install(TARGETS MomentumSlideSE RUNTIME DESTINATION SKSE/Plugins COMPONENT Slide)
install(FILES config/MomentumSlideSE.ini DESTINATION SKSE/Plugins COMPONENT Slide)
install(FILES README.md COMPATIBILITY.md VERIFICATION.md LICENSE EXCEPTIONS.md DESTINATION Docs/MomentumSlideSE COMPONENT Slide)
install(DIRECTORY licenses/ DESTINATION Docs/MomentumSlideSE/licenses COMPONENT Slide)

add_executable(SlideRuntimeTests tests/RuntimeSupportTests.cpp)
target_include_directories(SlideRuntimeTests PRIVATE src)
add_test(NAME RuntimeSupport COMMAND SlideRuntimeTests)
add_executable(SlideLayoutTests tests/RuntimeLayoutTests.cpp)
target_link_libraries(SlideLayoutTests PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(SlideLayoutTests PRIVATE ENABLE_COMMONLIBSSE_TESTING=1)
add_test(NAME RuntimeLayouts COMMAND SlideLayoutTests)
