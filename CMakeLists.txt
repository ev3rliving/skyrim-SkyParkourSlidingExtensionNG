cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0141)
    cmake_policy(SET CMP0141 NEW)
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
endif()
project(MomentumSlideSE VERSION 0.3.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use the x64 Microsoft C++ toolchain.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(COMMONLIB_PACKAGE "${CMAKE_CURRENT_SOURCE_DIR}/../SkyrimSE/package" CACHE PATH "Installed CommonLibSSE-NG package")
find_package(spdlog CONFIG REQUIRED PATHS "${COMMONLIB_PACKAGE}/lib/cmake/spdlog" NO_DEFAULT_PATH)
include("${COMMONLIB_PACKAGE}/lib/cmake/CommonLibSSE/CommonLibSSE-targets.cmake")
add_library(MomentumSlideSE SHARED src/main.cpp)
target_link_libraries(MomentumSlideSE PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(MomentumSlideSE PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MomentumSlideSE PRIVATE /W4 /permissive- /utf-8)
enable_testing()
add_executable(SlidePhysicsTests tests/SlidePhysicsTests.cpp)
target_include_directories(SlidePhysicsTests PRIVATE src)
add_test(NAME SlidePhysics COMMAND SlidePhysicsTests)
install(TARGETS MomentumSlideSE RUNTIME DESTINATION SKSE/Plugins)
install(FILES config/MomentumSlideSE.ini DESTINATION SKSE/Plugins)
install(FILES README.md VERIFICATION.md DESTINATION Docs/MomentumSlideSE)
install(DIRECTORY licenses/ DESTINATION Docs/MomentumSlideSE/licenses)
